function Show-ItemNavigator {
    param (
        [string]$StartPath = "C:\"
    )

    Write-Host "🌟 File Navigator By Dani 🌟`n"
    Write-Host "Ini adalah alat untuk menjelajahi dan mengelola file dan folder di komputer Anda.`n"
    Write-Host "Gunakan perintah untuk bernavigasi, melihat isi folder, atau menghapus file dan folder."
    Write-Host "Saat siap, tekan tombol Enter untuk melanjutkan...`n"
    $null = Read-Host "Tekan Enter untuk memulai"

    $history = @()
    $currentPath = $StartPath

    while ($true) {
        Clear-Host
        Write-Host "📂 Current Folder: $currentPath`n"

        # Ambil folder dan file, ukur ukuran
        $items = Get-ChildItem -Path $currentPath -ErrorAction SilentlyContinue | ForEach-Object {
            $size = 0
            if ($_.PSIsContainer) {
                $size = (Get-ChildItem $_.FullName -Recurse -File -ErrorAction SilentlyContinue |
                    Measure-Object -Property Length -Sum).Sum
            } else {
                $size = $_.Length
            }

            [PSCustomObject]@{
                Path = $_.FullName
                Name = $_.Name
                Type = if ($_.PSIsContainer) { "Folder" } else { "File" }
                SizeGB = [math]::Round($size / 1GB, 2)
                IsFolder = $_.PSIsContainer
            }
        } | Sort-Object SizeGB -Descending | Select-Object -First 10

        if (-not $items) {
            Write-Host "⚠️  No items found or access denied.`n"
        } else {
            $i = 1
            foreach ($item in $items) {
                Write-Host "$i. [$($item.Type)] $($item.Name) - $($item.SizeGB) GB"
                $i++
            }
        }

        Write-Host "`n📌 Commands:"
        Write-Host "   [1-10] Enter folder (if applicable)"
        Write-Host "   [1 del] Delete item with confirm"
        Write-Host "   [s] Show all items in folder"
        Write-Host "   [b] Back"
        Write-Host "   Press Ctrl+C to quit`n"

        $input = Read-Host "Enter command"

        switch ($input) {
            'b' {
                if ($history.Count -gt 0) {
                    $currentPath = $history[-1]
                    $history = $history[0..($history.Count - 2)]
                } else {
                    Write-Host "↩️  No previous folder."
                    Start-Sleep -Seconds 1.5
                }
            }
            's' {
                Clear-Host
                Write-Host "📁 Contents of: $currentPath`n"
                $allItems = Get-ChildItem -Path $currentPath -ErrorAction SilentlyContinue | Select-Object Name, Mode, Length, LastWriteTime
                if ($allItems.Count -eq 0) {
                    Write-Host "No files or folders found."
                } else {
                    $allItems | Format-Table -AutoSize
                }
                Write-Host "`nPress any key to continue..."
                $null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
            }
            default {
                if ($input -match '^(\d+)\s+del$') {
                    $index = [int]$matches[1] - 1
                    if ($index -ge 0 -and $index -lt $items.Count) {
                        $target = $items[$index]
                        Write-Host "`n⚠️ Are you sure you want to delete: $($target.Path) ? [y/n]"
                        $confirm = Read-Host
                        if ($confirm -eq 'y') {
                            try {
                                Remove-Item -Path $target.Path -Recurse -Force -ErrorAction Stop
                                Write-Host "✅ Deleted: $($target.Name)"
                            } catch {
                                Write-Host "❌ Failed to delete: $_"
                            }
                            Start-Sleep -Seconds 2
                        } else {
                            Write-Host "❎ Deletion cancelled."
                            Start-Sleep -Seconds 1.5
                        }
                    } else {
                        Write-Host "❌ Invalid item number."
                        Start-Sleep -Seconds 1.5
                    }
                }
                elseif ($input -match '^\d+$') {
                    $index = [int]$input - 1
                    if ($index -ge 0 -and $index -lt $items.Count) {
                        $selected = $items[$index]
                        if ($selected.IsFolder) {
                            $history += $currentPath
                            $currentPath = $selected.Path
                        } else {
                            Write-Host "📄 '$($selected.Name)' is a file, not a folder."
                            Start-Sleep -Seconds 1.5
                        }
                    } else {
                        Write-Host "❌ Invalid number."
                        Start-Sleep -Seconds 1.5
                    }
                } else {
                    Write-Host "❌ Invalid command."
                    Start-Sleep -Seconds 1.5
                }
            }
        }
    }
}

# Jalankan fungsi
Show-ItemNavigator
